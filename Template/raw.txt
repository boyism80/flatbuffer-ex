{{~
func raw_type(field)
    if field.is_array
        ret '[' + raw_type(field.array_element) + ']'
    else if field.is_nullable_scalar
        ret 'nullable.' + field.nullable_table_name
    else
        ret raw_name(field)
    end
end
~}}

{
{{~ for value in enum.values ~}}
    {{ value.name }}{{ if value.value }} = {{ value.value }}{{ end }}{{ if !for.last }},{{ end }}
{{~ end ~}}
}

{{~ end ~}}
{{~ end ~}}
{{~ end ~}}
